# NetSurf Build System for Black OS
# This file defines the build configuration for NetSurf integration

NETSURF_ROOT := $(BLACKOS_ROOT)/netsurf

# NetSurf source files to compile
NETSURF_SOURCES := \
    $(NETSURF_ROOT)/src/browsers.c \
    $(NETSURF_ROOT)/src/layout/layout.c \
    $(NETSURF_ROOT)/src/parser/parser.c \
    $(NETSURF_ROOT)/src/http/http.c \
    $(NETSURF_ROOT)/src/utf8/utf8.c \
    $(NETSURF_ROOT)/utils/brush/brush.c

# Black OS browser sources
BROWSER_SOURCES := \
    src/kernel/browser.c \
    src/kernel/win_browser.c \
    src/kernel/app_browser.c

# All browser-related sources
ALL_BROWSER_SOURCES := $(BROWSER_SOURCES) $(NETSURF_SOURCES)

# NetSurf compiler flags
NETSURF_CFLAGS := -D NETSURF_EMBEDDED \
    -D NO_JAVASCRIPT \
    -D NO_THREADS \
    -D NS_HAVE_DIRTY_RENDERING=0 \
    -D NS_HAVE_DRAG_N_DROP=0 \
    -D NS_HAVE_MOUSE_CURSOR=0 \
    -D NS_HAVE_SPRITES=0 \
    -D NS_HAVE_ICONS=0 \
    -D NS_HAVE_PLUGINS=0 \
    -D NS_HAVE_BENCHMARKS=0 \
    -D NS_HAVE_DEBUG=0

# Include directories
NETSURF_INCLUDES := -I$(NETSURF_ROOT)/include -I$(NETSURF_ROOT)/src

# Compile browser sources
$(BUILD)/kernel/browser/%.o: $(1)
	@mkdir -p $(dir $@)
	$(CC) $(CFLAGS) $(NETSURF_CFLAGS) $(NETSURF_INCLUDES) -c $< -o $@

# NetSurf integration complete markers
.PHONY: browser-neturf browser-verify

browser-neturf: $(ALL_BROWSER_SOURCES)
	@echo "NetSurf browser integration ready"

browser-verify:
	@echo "Verifying NetSurf browser integration..."
	@grep -q "NetSurf" src/kernel/browser.c && echo "✅ NetSurf integration verified"
	@grep -q "browser_create_window" src/kernel/browser.c && echo "✅ Browser window creation verified"
	@grep -q "browser_navigate_url" src/kernel/browser.c && echo "✅ Browser navigation verified"
